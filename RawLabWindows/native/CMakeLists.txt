cmake_minimum_required(VERSION 3.24)
project(RawLabWindows LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
include(CTest)
add_compile_options(/utf-8)
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(zlib
    URL https://github.com/madler/zlib/archive/refs/tags/v1.3.1.tar.gz
    URL_HASH SHA256=17e88863f3600672ab49182f217281b6fc4d3c762bde361935e436a95214d05c)
FetchContent_MakeAvailable(zlib)
add_library(ZLIB::ZLIB ALIAS zlibstatic)
set_property(TARGET zlibstatic PROPERTY INTERFACE_INCLUDE_DIRECTORIES
    "$<BUILD_INTERFACE:${zlib_SOURCE_DIR}>;$<BUILD_INTERFACE:${zlib_BINARY_DIR}>;$<INSTALL_INTERFACE:include>")
install(TARGETS zlibstatic EXPORT OpenCVModules ARCHIVE DESTINATION lib)
set(ZLIB_LIBRARY zlibstatic CACHE STRING "Target zlib library" FORCE)
set(ZLIB_INCLUDE_DIR "${zlib_SOURCE_DIR}" CACHE PATH "Target zlib headers" FORCE)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${REPO}/RawLabWindows/Rendering/HDR" "${CMAKE_BINARY_DIR}/hdr-display-build")
include("${REPO}/lutools/cmake/libraw/libraw.cmake")
rawlab_add_libraw_shared()
target_compile_definitions(raw PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX NOGDI)
target_compile_definitions(raw PRIVATE LIBRAW_BUILDLIB _CRT_SECURE_NO_WARNINGS NOMINMAX)
target_compile_options(raw PRIVATE /utf-8 /O2 /openmp)
target_link_libraries(raw PRIVATE ZLIB::ZLIB ws2_32)
set(SONY2FUJI_ENABLE_DNGSDK ON CACHE BOOL "" FORCE)
set(BUILD_CLI ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIB ON CACHE BOOL "" FORCE)
set(SONY2FUJI_ENABLE_GPU ON CACHE BOOL "" FORCE)
set(SONY2FUJI_ENABLE_X3F ON CACHE BOOL "" FORCE)
set(RAWLAB_X3F_RUST_TARGET x86_64-pc-windows-msvc CACHE STRING "" FORCE)
set(SONY2FUJI_ENABLE_LENSFUN ON CACHE BOOL "" FORCE)
set(SONY2FUJI_BUILD_LENS_DEPS ON CACHE BOOL "" FORCE)
# MSVC's classic OpenMP lacks the canonical unsigned loops used by the core.
set(SONY2FUJI_ENABLE_OPENMP OFF CACHE BOOL "" FORCE)
set(SONY2FUJI_ENABLE_WAVELET_DENOISE ON CACHE BOOL "" FORCE)
set(SONY2FUJI_ENABLE_CHROMA_DENOISE ON CACHE BOOL "" FORCE)
set(SONY2FUJI_BUILD_DENOISE_DEPS ON CACHE BOOL "" FORCE)
add_subdirectory("${REPO}/lutools" core)
set_target_properties(sony2fuji_cli PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/cli")
# The RAW conversion uses signed row loops compatible with classic MSVC OpenMP.
set_property(SOURCE "${REPO}/lutools/src/core/raw_processor.cpp" DIRECTORY "${REPO}/lutools"
    APPEND PROPERTY COMPILE_OPTIONS /openmp)
target_sources(sony2fuji PRIVATE thumbnail.cpp)
if(BUILD_TESTING)
    add_executable(windows_display_image_tests ../tests/native/display_image_checks.cpp)
    target_include_directories(windows_display_image_tests PRIVATE "${REPO}/lutools/include")
    target_link_libraries(windows_display_image_tests PRIVATE sony2fuji ZLIB::ZLIB TIFF::tiff windowscodecs ole32)
    set_target_properties(windows_display_image_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    add_test(NAME windows_display_image COMMAND windows_display_image_tests "${CMAKE_BINARY_DIR}/display-image-fixtures")
    set_tests_properties(windows_display_image PROPERTIES LABELS "windows;ci_windows")
    add_executable(windows_denoise_probe EXCLUDE_FROM_ALL ../tests/native/denoise_probe.cpp)
    target_include_directories(windows_denoise_probe PRIVATE "${REPO}/lutools/include" "${REPO}/lutools/src")
    target_compile_definitions(windows_denoise_probe PRIVATE NOMINMAX SONY2FUJI_ENABLE_D3D11)
    target_link_libraries(windows_denoise_probe PRIVATE sony2fuji_core)
    set_target_properties(windows_denoise_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/licenses")
configure_file("${REPO}/RawLabMac/Resources/Licenses/LibRaw-X3F-BSD.txt"
    "${CMAKE_BINARY_DIR}/licenses/LibRaw-X3F-BSD.txt" COPYONLY)
foreach(name IN LISTS RAWLAB_LIBRAW_LICENSE_FILES)
    configure_file("${LIBRAW_SOURCE_DIR}/${name}" "${CMAKE_BINARY_DIR}/licenses/LibRaw-${name}" COPYONLY)
endforeach()
foreach(name README.md LICENSE.LGPL-2.1)
    configure_file("${REPO}/lutools/third_party/nikon_he/${name}"
        "${CMAKE_BINARY_DIR}/licenses/nikon-he/${name}" COPYONLY)
endforeach()
configure_file("${zlib_SOURCE_DIR}/LICENSE" "${CMAKE_BINARY_DIR}/licenses/zlib-LICENSE" COPYONLY)
configure_file("${REPO}/lutools/third_party/nlohmann/LICENSE.MIT"
    "${CMAKE_BINARY_DIR}/licenses/nlohmann-JSON-MIT.txt" COPYONLY)
