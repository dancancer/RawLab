cmake_minimum_required(VERSION 3.24)
project(RawLabWindows LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
include(CTest)
add_compile_options(/utf-8)
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(zlib
    URL https://github.com/madler/zlib/archive/refs/tags/v1.3.1.tar.gz
    URL_HASH SHA256=17e88863f3600672ab49182f217281b6fc4d3c762bde361935e436a95214d05c)
FetchContent_MakeAvailable(zlib)
add_library(ZLIB::ZLIB ALIAS zlibstatic)
target_include_directories(zlibstatic PUBLIC "${zlib_SOURCE_DIR}" "${zlib_BINARY_DIR}")
FetchContent_Declare(libraw
    DOWNLOAD_EXTRACT_TIMESTAMP FALSE
    URL https://www.libraw.org/data/LibRaw-0.22.2.tar.gz
    URL_HASH SHA256=de86b035655accff8d4010f1a221fdf50d353cb7b1422ba26f14a0db92612cfa)
FetchContent_MakeAvailable(libraw)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LIBRAW_SOURCE_DIR "${libraw_SOURCE_DIR}")
include("${REPO}/lutools/cmake/patch-libraw.cmake")
file(GLOB_RECURSE LIBRAW_SOURCES CONFIGURE_DEPENDS "${libraw_SOURCE_DIR}/src/*.cpp")
list(FILTER LIBRAW_SOURCES EXCLUDE REGEX "_ph\\.cpp$")
add_library(raw SHARED ${LIBRAW_SOURCES})
target_include_directories(raw PUBLIC "${libraw_SOURCE_DIR}")
target_compile_definitions(raw PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX NOGDI)
target_compile_definitions(raw PRIVATE LIBRAW_BUILDLIB USE_ZLIB _CRT_SECURE_NO_WARNINGS NOMINMAX)
target_compile_options(raw PRIVATE /utf-8 /O2 /openmp)
target_link_libraries(raw PRIVATE ZLIB::ZLIB ws2_32)
set(SONY2FUJI_LIBRAW_TARGET raw)
set(BUILD_CLI OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIB ON CACHE BOOL "" FORCE)
set(SONY2FUJI_ENABLE_GPU ON CACHE BOOL "" FORCE)
# MSVC's classic OpenMP lacks the canonical unsigned loops used by the core.
set(SONY2FUJI_ENABLE_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory("${REPO}/lutools" core)
# The RAW conversion uses signed row loops compatible with classic MSVC OpenMP.
set_property(SOURCE "${REPO}/lutools/src/core/raw_processor.cpp" DIRECTORY "${REPO}/lutools"
    APPEND PROPERTY COMPILE_OPTIONS /openmp)
target_sources(sony2fuji PRIVATE thumbnail.cpp)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/licenses")
foreach(name COPYRIGHT LICENSE.CDDL LICENSE.LGPL)
    configure_file("${libraw_SOURCE_DIR}/${name}" "${CMAKE_BINARY_DIR}/licenses/LibRaw-${name}" COPYONLY)
endforeach()
configure_file("${zlib_SOURCE_DIR}/LICENSE" "${CMAKE_BINARY_DIR}/licenses/zlib-LICENSE" COPYONLY)
