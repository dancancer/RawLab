cmake_minimum_required(VERSION 3.15)
project(rawlab_wavelib LANGUAGES C)
if(NOT WAVELIB_SOURCE_DIR)
    message(FATAL_ERROR "WAVELIB_SOURCE_DIR is required")
endif()
file(READ "${WAVELIB_SOURCE_DIR}/src/wavelib.c" source)
string(REPLACE "\r\n" "\n" source "${source}")
file(READ "${CMAKE_CURRENT_LIST_DIR}/../wavelib-swt-columns.c.inc" swt_columns)
string(REPLACE "double* swt2(" "${swt_columns}\ndouble* swt2(" source "${source}")
foreach(band lp hp)
    if(band STREQUAL "lp")
        set(low aLL)
        set(high aLH)
    else()
        set(low aHL)
        set(high aHH)
    endif()
    set(old "for (i = 0; i < ic; ++i) {\n\t\t\tswt_per_stride(M,${band}_dn1 + i, ir, wt->wave->lpd, wt->wave->hpd, lp, wavecoeff + ${low} + i, rows_i, wavecoeff + ${high} + i, istride, ostride);\n\t\t}")
    string(FIND "${source}" "${old}" found)
    if(found LESS 0)
        message(FATAL_ERROR "Unexpected pinned wavelib SWT column layout")
    endif()
    string(REPLACE "${old}" "rawlab_swt_columns(M, ${band}_dn1, ir, ic, wt->wave->lpd, wt->wave->hpd, lp, wavecoeff + ${low}, wavecoeff + ${high});" source "${source}")
endforeach()
string(FIND "${source}" "void iswt2(" begin)
string(FIND "${source}" "double* modwt2(" end)
if(begin LESS 0 OR end LESS begin)
    message(FATAL_ERROR "Unexpected pinned wavelib source layout")
endif()
string(SUBSTRING "${source}" 0 ${begin} before)
string(SUBSTRING "${source}" ${end} -1 after)
file(READ "${CMAKE_CURRENT_LIST_DIR}/../wavelib-iswt2.c.inc" replacement)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/wavelib.c" "${before}${replacement}\n${after}")
set(sources "${CMAKE_CURRENT_BINARY_DIR}/wavelib.c")
foreach(name conv cwt cwtmath hsfft real wavefilt wavefunc wtmath)
    list(APPEND sources "${WAVELIB_SOURCE_DIR}/src/${name}.c")
endforeach()
add_library(wavelib STATIC ${sources})
set_target_properties(wavelib PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(wavelib PUBLIC "${WAVELIB_SOURCE_DIR}/header" PRIVATE "${WAVELIB_SOURCE_DIR}/src")
if(NOT MSVC)
    target_link_libraries(wavelib PUBLIC m)
endif()
install(TARGETS wavelib ARCHIVE DESTINATION lib)
install(FILES "${WAVELIB_SOURCE_DIR}/header/wavelib.h" DESTINATION include)
install(FILES "${WAVELIB_SOURCE_DIR}/COPYRIGHT" DESTINATION share/licenses/wavelib)
configure_file(wavelib.pc.in wavelib.pc @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/wavelib.pc" DESTINATION lib/pkgconfig)
